package core

import (
	"context"

	"github.com/google/uuid"
)

func skewChannel(on bool) Channel {
	return Channel{Principal: "skew:principal", SourceOpeningBinding: on}
}

func skewExpand(s *Store, req ExpandRequest, dest Destination, declared bool) (Expansion, error) {
	return s.ExpandForReader(context.Background(), req, dest, declared)
}

func skewPull(receipt, handle string, span *ByteSpanRequest, opening int) ExpandRequest {
	return ExpandRequest{RequestID: uuid.NewString(), ReceiptID: receipt, Handle: handle, Span: span, OpeningBytes: opening}
}
